Table paie_config_legale {
  id_config_legale integer [pk, increment]
  code_version varchar(30) [unique, not null, note: 'ex: LF_2026, LF_2027']
  libelle varchar(200) [not null]
  date_debut date [not null]
  date_fin date [note: 'NULL = en vigueur']
  
  // Salaire Minimum & Plafonds Sociaux
  sme_national decimal(12,2) [not null, note: 'SME national non agricole, ex: 262680']
  sme_agricole decimal(12,2) [not null, note: 'SME secteur agricole, ex: 266500']
  plafond_cotisation_nb_sme decimal(4,2) [default: 8.00, note: 'Multiplicateur plafond CNaPS / Santé (8 x SME = 2 101 440 Ar)']
  
  // Fiscalité IRSA
  seuil_exoneration_irsa decimal(12,2) [not null, note: 'Ex: 350 000 Ar']
  minimum_perception_irsa decimal(12,2) [not null, note: 'Ex: 3 000 Ar en 2026']
  deduction_charge_famille decimal(12,2) [not null, note: 'Ex: 2 000 Ar / enfant']
  arrondi_base_irsa varchar(20) [default: 'CENTAINE_INF']
  
  // Temps de Travail (Loi 2003-044)
  duree_hebdo_legale decimal(5,2) [default: 40.00]
  diviseur_mensuel_heures decimal(7,4) [default: 173.3333, note: '(40 x 52) / 12']
  seuil_hs_tranche1 decimal(5,2) [default: 8.00, note: '8 premières HS à +30%']
  plafond_hs_hebdo decimal(5,2) [default: 20.00, note: 'Max légal hebdo']
  
  // Congés Payés (Art. 85)
  acquisition_conge_mensuel_jours decimal(4,2) [default: 2.50, note: '2.5 j/mois = 30 j/an']
  diviseur_valorisation_conge_jours integer [default: 30, note: 'Salaire brut ÷ 30']

  Note: 'SP4 - Millésime légal évolutif pour la Loi de Finances et les décrets SME/Plafonds.'
}

Table ref_type_cotisation {
  id_type_cotisation integer [pk, increment]
  id_config_legale integer [ref: > paie_config_legale.id_config_legale, note: 'Millésime associé']
  code varchar(20) [unique, not null, note: 'ex: CNAPS, OSTIE, OMSI, ISI']
  libelle varchar(100) [not null]
  taux_salarial decimal(5,2) [note: 'ex: 1.00 pour OSTIE, 1.50 pour OMSI']
  taux_patronal decimal(5,2) [note: 'ex: 5.00 pour OSTIE, 5.50 pour OMSI']
  est_plafonee boolean [default: true, note: 'Plafonné à 8x SME']
  date_debut date [not null, note: 'AJOUTE — historisation, un taux change dans le temps']
  date_fin date [note: 'AJOUTE — NULL = taux actuellement en vigueur']
}

Table bareme_irsa {

  tranche_min decimal(12,2) [not null]
  tranche_max decimal(12,2) [note: 'NULL = dernière tranche, sans plafond']
  taux decimal(5,2) [not null, note: 'ex: 5.00 pour 5%']
  date_debut date [not null]
  date_fin date [note: 'NULL = barème actuellement en vigueur']

  Note: 'AJOUTE — l\'IRSA est un barème progressif à tranches, ref_type_cotisation (taux unique) ne peut pas le représenter'

Table employe_profil_pointage {
  id_profil integer [pk, increment]
  id_employe integer [ref: > employe.id_employe, not null]

  statut_pointage varchar(20) [not null, default: 'OBLIGATOIRE', note: 'OBLIGATOIRE, FACULTATIF, EXEMPTE']

  date_fin date [note: 'NULL = permanent ; renseigné si la dérogation est à durée déterminée (ex: intérim, détachement temporaire)']
  motif text

  created_at timestamp
  updated_at timestamp

  Note: 'SCINDE de l\'ancienne table employe_profil_pointage_indemnite — ne porte QUE le statut de pointage. Indépendant du droit à indemnité (cf. cas du double emploi : pointage obligatoire maintenu, indemnité retirée).'
}

Table employe_indemnite_derogation {
  id_derogation integer [pk, increment]
  id_employe integer [ref: > employe.id_employe, not null]
  id_indemnite integer [ref: > ref_indemnite.id_indemnite, not null]

  beneficie boolean [not null, note: 'true = accorde une indemnité que le site ne donne pas par défaut (ex: DAF + Repas) ; false = la retire (ex: double emploi)']

  date_debut date [not null]
  date_fin date [note: 'NULL = dérogation permanente']
  motif text [note: 'ex: "Double emploi — activité principale hors GALAXAM"']

  created_at timestamp

  Note: 'NIVEAU 2 de la cascade indemnité. Généralisée via ref_indemnite (remplace les anciens booléens fixes droit_indemnite_repas/deplacement) — cohérent avec l\'existence du référentiel ref_indemnite.'
}

Table pointage_journalier {
  id_pointage integer [pk, increment]
  id_employe integer [ref: > employe.id_employe, not null]
  id_affectation integer [ref: > employe_affectation_carriere.id_affectation, not null]

  date_pointage date [not null]

  heure_arrivee_matin time
  heure_sortie_matin time
  heure_arrivee_apres_midi time
  heure_sortie_apres_midi time

  heures_travaillees decimal(4,2) [default: 0.00]
  heures_normales decimal(4,2) [default: 0.00]
  heures_supplementaires decimal(4,2) [default: 0.00, note: 'Valeur BRUTE journalière — recalculée/compensée au niveau mensuel dans pointage_synthese_mensuelle, ne pas l\'utiliser telle quelle pour la paie']

  statut_presence varchar(20) [default: 'PRESENT', note: 'PRESENT, ABSENT, CONGE, FERIE, EXEMPTE_PRESENT']

  avec_indemnite_repas boolean [default: false, note: 'NIVEAU 3 de la cascade — pré-coché selon site+dérogation, modifiable manuellement (case à cocher)']
  avec_indemnite_deplacement boolean [default: false, note: 'idem — décoché manuellement si venue en véhicule personnel ce jour']
  avec_second_repas boolean [default: false, note: 'accordé si heure_sortie_apres_midi >= parametre_global(SEUIL_SECOND_REPAS), et seulement si avec_indemnite_repas = true']
  motif_retrait_indemnite varchar(100) [note: 'AJOUTE — optionnel, renseigné uniquement quand une case est décochée/cochée manuellement à l\'encontre de la règle par défaut. Remplace mode_venue : la case à cocher suffit, ce champ ne sert qu\'à justifier une exception, pas à la calculer.']

  origine_saisie varchar(20) [default: 'MANUEL', note: 'MANUEL, IMPORT, AUTOMATIQUE_EXEMPTE']
  est_modifie_manuellement boolean [default: false]

  created_at timestamp
  updated_at timestamp
}

Table pointage_synthese_mensuelle {
  id_synthese integer [pk, increment]
  id_employe integer [ref: > employe.id_employe, not null]
  id_affectation integer [ref: > employe_affectation_carriere.id_affectation, not null]
  mois integer [not null]
  annee integer [not null]

  total_heures_travaillees decimal(6,2) [not null]
  total_retard_cumule decimal(6,2) [not null, default: 0]
  total_surplus_cumule decimal(6,2) [not null, default: 0]
  solde_net decimal(6,2) [not null, note: 'positif = heures sup à payer, négatif = absence à déduire — compensation retard/surplus au niveau mensuel']
  heures_sup_validees decimal(6,2) [not null, default: 0, note: 'Forcé à 0 si employe_affectation_carriere.categorie_prof = C (Cadre), quel que soit le solde_net']
  absence_valorisee decimal(6,2) [not null, default: 0]

  statut_calcul varchar(20) [default: 'BROUILLON', note: 'BROUILLON, VALIDE, INTEGRE_PAIE']
  id_utilisateur_validation integer
  date_validation timestamp

  created_at timestamp
  updated_at timestamp

  Note: 'AJOUTE — agrège pointage_journalier (saisie brute quotidienne) en un solde mensuel net, seul niveau où s\'applique la règle de compensation retard/heures sup et le blocage Cadre.'
}
